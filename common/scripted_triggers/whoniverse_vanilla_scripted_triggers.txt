has_energy_upkeep = {
	OR = {
		is_robotic_or_voidspawn_species = yes
		has_trait = trait_cyborg_power_intensive
		has_trait = trait_exd
		species = { has_trait = trait_whoniverse_auton } # modded in
	}
}

		is_archetype = BIOLOGICAL

	}
}
