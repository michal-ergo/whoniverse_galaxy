portraits = {
	# Sontaran Classic
	wise_sontaran_classic_drone = {	
		entity = "portrait_sontaran_classic"
		clothes_selector = "no_texture" 
		attachment_selector = "no_texture" 
		greeting_sound = "whoniverse_sontaran_classic_greetings"
		character_textures = {	
			"gfx/models/portraits/wise_sontaran/classic/sontaran_classic_drone.dds"
			"gfx/models/portraits/wise_sontaran/classic/sontaran_classic_helmet.dds"
			}
		}
	}

portrait_groups = {
	
	wise_sontaran_classic = {
		default = wise_sontaran_classic_drone
		game_setup = { #will run with a limited country scope. species and government is set but the country does not actually exist
			add = {
				portraits = {
					wise_sontaran_classic_drone
				}
			}
		}		
		
		#species scope
		species = { #generic portrait for a species
			set = {
				portraits = {
					wise_sontaran_classic_drone
				}
			}
		}		
		
		#pop scope
		pop = { #for a specific pop
			add = {
				portraits = {
					wise_sontaran_classic_drone
				}
			}
		}
		
		#leader scope
		leader = { #scientists, generals, admirals, governor
			
			set = {
				trigger = {
					leader_class = commander
				}
				portraits = {
					wise_sontaran_classic_drone
				}
			}
			
			add = {
				trigger = {
					leader_class = scientist
				}
				portraits = {
					wise_sontaran_classic_drone
				}
			}
			
			add = {
				trigger = {
					leader_class = official
				}
				portraits = {
					wise_sontaran_classic_drone
				}
			}
		}		
	}
}
