# Zygor anomaly
ship_event = {
    id = whoniverse.260
    title = "whoniverse.260.name"
    desc = "whoniverse.260.desc"
    picture = GFX_evt_alien_planet
	location = from
	show_sound = event_wind_ruins
	fire_only_once = yes
	is_triggered_only = yes

	trigger = {
		owner = {
			is_ai = no
		}
		solar_system = { has_star_flag = whoniverse_zygor_system }
		FROM = {
			has_planet_flag = whoniverse_zygor
		}
	}

	immediate = {
		owner = { set_country_flag = whoniverse_zygor_archaeology }
		from = { prevent_anomaly = yes }
	}

	option = {
		name = CURIOUS
		from = {
			create_archaeological_site = whoniverse_digsite_zygor
			save_global_event_target_as = zygor_event
		} 
		owner = {
			add_monthly_resource_mult = {
				resource = society_research
				value = @tier1researchreward
				min = @tier1researchmin
				max = @tier1researchmax
			}
		}
	}
}

					count = 3

						species = event_target:zygon_species

		}
	}

# Zygor project successful - either creates new Zygon species, or adds extra Zygons if they exist already
country_event = {
	id = whoniverse.101
	title = "PROJECT_COMPLETE"
	desc = "whoniverse.101.desc"
	picture = GFX_evt_whoniverse_zygon
	show_sound = whoniverse_zygon_event
	location = ROOT
	fire_only_once = yes
		
	is_triggered_only = yes
		
	option = {
		name = whoniverse.101.a
		custom_tooltip = whoniverse.101.tooltip
		hidden_effect = {
			if = {
				limit = { exists = event_target:zygon_species }
				while = {
					count = 2
					capital_scope = {
						create_pop = {
							species = event_target:zygon_species
							}
						}
					}
			}
			else_if = {
				limit = { NOT = { exists = event_target:zygon_species } }
			capital_scope = {
				create_species = {
					name = "WHO_ZYGON"
					class = FUN
					portrait = whoniverse_zygon
					traits = {
						# trait = "trait_pc_tropical_preference"
						trait = "trait_whoniverse_zygon"
						trait = "trait_whoniverse_bio_electricity"
						trait = "trait_adaptive"
						trait = "trait_repugnant"
						trait = "trait_deviants"
						}
					homeworld = root
					effect = {
						save_global_event_target_as = zygon_species
						}
					}
				while = {
					count = 2
					create_pop_group = {
						species = event_target:zygon_species
						effect = { save_event_target_as = zygon_pop_group }
					}	
				}			
				event_target:zygon_pop_group = { 
					set_citizenship_type = {
						type = citizenship_limited
						cooldown = no
						}
					}
				}
			}
		}
	}
}
