# Roboform capital event 1
planet_event = {
	id = whoniverse.351
	title = "whoniverse.351.name"
	desc = "whoniverse.351.desc"
	picture = GFX_evt_whoniverse_roboform
	show_sound = whoniverse_roboform_music
	location = ROOT
	trackable = yes
	fire_only_once = yes
	is_triggered_only = yes

	pre_triggers = {
		has_owner = yes
		is_ai = no
		is_occupied_flag = no
	}

	trigger = {
		num_pops > 0
	}

	immediate = {
		if = {
			limit = { NOT = { exists = event_target:roboform_species } }
			create_species = {
					name = "WHO_ROBOFORM"
					class = ROBOT
					namelist = MACHINE1
					portrait = whoniverse_roboform
					traits = {
						trait = trait_wet_planet_preference
						trait = trait_auto_mod_robotic
						trait = trait_robot_uncanny
						}
					homeworld = root
					effect = {
						save_global_event_target_as = roboform_species
						}
					}			
			}
		}

	option = {
		name = whoniverse.351.a
		custom_tooltip = whoniverse.351.a.tooltip
		trigger = {
			# owner = { resource_stockpile_compare = { resource = energy value >= 500 } } 
			NOT = { owner = { has_authority = auth_hive_mind } }
			owner = { is_whoniverse_dalek = no }
		}
		# owner = {
		# 	add_resource = { energy = -500 }
		# }
		hidden_effect = {
			while = {
				count = 2
					create_pop = {
						species = event_target:roboform_species
					}
				}
		}
	}

	option = {
		name = whoniverse.351.b
		custom_tooltip = whoniverse.351.b.tooltip
		trigger = {
			NOT = { owner = { has_authority = auth_hive_mind } }
		}
		this = {
			save_event_target_as = roboform_planet
		}
		owner = {
			create_fleet = {
				name = "ROBOFORM_FLEET"
				effect = {
					set_owner = prev.owner
					set_location = {
						target = event_target:roboform_planet 
						distance = 10
						angle = random
						}
					while = {
						count = 3
						create_army_transport = {
							ship_name = "ROBOFORM_SHUTTLE"
							army_name = "ROBOFORM_ARMY"
							army_type = "robotic_army"
							species = event_target:roboform_species
							}
						}
					}
				}
			}
		}

	option = {
		trigger = {
			owner = { is_whoniverse_dalek = no }
		}
		name = whoniverse.351.c
		# custom_tooltip = whoniverse.351.a.tooltip
	}

	option = {
		name = whoniverse.351.d
		# custom_tooltip = whoniverse.351.a.tooltip
		trigger = {
			owner = { is_whoniverse_dalek = yes	}
		}
		owner = {
			add_monthly_resource_mult = {
				resource = influence
				value = @tier1influencereward
				min = @tier1influencemin
				max = @tier1influencemax
			}
		}
	}

	option = {
		name = whoniverse.351.e
		custom_tooltip = whoniverse.351.e.tooltip
		trigger = {
			owner = { has_whoniverse_cyber_conversion = yes }
		}
		hidden_effect = {
			while = {
				count = 2
				create_pop = {
					species = owner_main_species
				}
			}
		}
	}
