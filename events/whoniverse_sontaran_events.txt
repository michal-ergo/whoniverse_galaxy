	id = whoniverse.384
	hide_window = yes
	is_triggered_only = yes
	fire_only_once = yes
  
	pre_triggers = {
		has_owner = yes
	}

	trigger = { 
		from = { has_origin = origin_whoniverse_sontaran } 
		has_planet_flag = whoniverse_rutan_empire_planet
	}

	immediate = {
		
		if = {
			limit = {
				fromfrom = {
					has_country_flag = whoniverse_rutan_empire_flag
				}
			}
			from = { country_event = { id = whoniverse.385 days = 1 random = 0}  }
		}
		else = {
			from = { country_event = { id = whoniverse.386 days = 1 random = 0}  }
		}
	}
}
